# 🌸 គេហទំព័រលិខិតអញ្ជើញបែបខ្មែរឌីជីថល (Khmer Royal E-Invitation)

គេហទំព័រ **E-Invitation បែបខ្មែរ** រចនាបទក្បាច់ខ្មែរប្រណិត ពណ៌មាស និងក្រហមទុំ (Royal Gold & Deep Burgundy) ជាមួយប្រព័ន្ធអន្តរកម្មទំនើប (Interactive Experience) សម្រាប់ពិធីមង្គលការ ឬកម្មវិធីប្រពៃណីខ្មែរ។

## 🌟 លក្ខណៈពិសេសៗ (Key Features)

1. **បទពិសោធន៍បើកស្រោមសំបុត្របែប 3D (Interactive 3D Envelope Unseal):**
   - មានត្រាមាសខ្មែរ 3D (*Royal Gold Lotus Seal*)
   - នៅពេលចុចលើត្រា នឹងមានសំឡេងរោទ៍កណ្ដឹងមាស ត្រាបែកផ្ទុះពន្លឺ (*Seal Break Explosion*) ចលនាបើកសំបុត្រ 3D និងបាញ់ផ្កាភ្លើងផ្កាឈូកមាស (*Grand Confetti Cannon*) យ៉ាងអស្ចារ្យ
2. **ចលនាបង្ហាញខ្លឹមសារតាម Scroll (Cinematic Scroll Reveal):**
   - ផ្នែកនីមួយៗ (Hero, Countdown, មាតាបិតា, កាលវិភាគ, ទីតាំង, RSVP) លេចឡើងមួយៗតាមលំដាប់លំដោយរលូន (Staggered Animation) ពេលភ្ញៀវអូសចុះក្រោម
3. **ចលនាអបអរសាទរពេលចុះឈ្មោះចូលរួម (RSVP Join Celebration):**
   - នៅពេលភ្ញៀវជ្រើសរើស "ខ្ញុំនឹងចូលរួម" ហើយចុច Submit នោះប្រព័ន្ធនឹងបាញ់ Confetti មាសរំលេចពេញអេក្រង់ជាការអបអរសាទរ
4. **ចលនាពន្លឺមាស និង 3D Tilt (Shimmer & Tilt Micro-interactions):**
   - ពន្លឺមាសរត់កាត់សន្លឹកបៀរ (*Gold Shimmer Sweep*)
   - ចលនារំញ័រ 3D តាមការរំកិលកណ្ដុរ ឬប៉ះអេក្រង់លើរូបថតគូស្វាមីភរិយា និងកាតភ្ញៀវ
   - ចលនារលកទឹក (*Ripple Wave Effect*) ពេលចុចលើគ្រប់ប៊ូតុងទាំងអស់
5. **តន្ត្រីកំដរមង្គលការ (TikTok Background Music):**
   - បានទាញយក និងបំពាក់បទចម្រៀងកំដរមង្គលការពី TikTok វីដេអូដែលអ្នកបានផ្ញើមក (`assets/audio/bg_music.mp3`)
   - មានប្រព័ន្ធចាក់ដោយស្វ័យប្រវត្តិតាមធម្មជាតិ (*Smooth Audio Fade-In*) និង Loop ជាប់ជាប្រចាំ
   - ប៊ូតុងបិទ/បើកតន្ត្រីអណ្តែតលើអេក្រង់ (*Floating Vinyl Disc*)
   - មានប្រព័ន្ធ Fallback ទៅកាន់ Web Audio Synthesizer ដោយស្វ័យប្រវត្តិប្រសិនបើគ្មាន Internet ឬ File mp3 មានបញ្ហា
3. **ការកំណត់ឈ្មោះភ្ញៀវផ្ទាល់ខ្លួនតាម VIP Link (Admin Only via /generate):**
   - សម្រាប់ភ្ញៀវទូទៅ ផ្នែកបង្កើត Link នេះនឹង **មិនបង្ហាញ** ឡើយ ដើម្បីរក្សាភាពផ្លូវការ
   - សម្រាប់ម្ចាស់ដើមការ (Admin) គ្រាន់តែចូលទៅកាន់៖
     👉 **`http://domain.com/generate`** (ឬ **`http://domain.com/?generate`**)
   - ម្ចាស់ដើមការអាចវាយបញ្ចូលឈ្មោះភ្ញៀវ (ឧ. `ឯកឧត្តម សុខ សាន`) រួចចុច **«ចម្លង Link»**

   - មានប៊ូតុង **«👁️ បើកមើលសំបុត្រ»**, **«📋 ចម្លង Link»**, និង **«✈️ ផ្ញើតាម Telegram»** ភ្លាមៗ

4. **នាឡិការាប់ថយក្រោយ (Live Countdown Timer):**
   - រាប់ថយក្រោយដល់ថ្ងៃកម្មវិធី (ថ្ងៃ ម៉ោង នាទី វិនាទី)

6. **ទីតាំង និងផែនទី (Google Maps Navigation):**

```sql
-- 1. Invitations table (multi-tenant parent record)
create table if not exists public.invitations (
  id uuid default gen_random_uuid() primary key,
  user_id uuid,
  slug text unique,
  title text,
  couple_names_khmer text,
  couple_names_en text,
  event_type text default 'wedding',
  event_date timestamp with time zone,
  venue_name text,
  venue_address text,
  map_url text,
  music_url text,
  music_playlist jsonb default '[]'::jsonb,
  cover_url text,
  story_text text,
  theme_config jsonb,
  is_active boolean default true,
  is_suspended boolean default false,
  view_count integer default 0,
  created_at timestamp with time zone default timezone('utc'::text, now()),
  updated_at timestamp with time zone default timezone('utc'::text, now())
);

-- Ensure the music playlist column exists on already-created tables
alter table public.invitations add column if not exists music_playlist jsonb default '[]'::jsonb;

-- 2. Create RSVPs table (linked to an invitation)
create table if not exists public.rsvps (
  id uuid default gen_random_uuid() primary key,
  invitation_id uuid references public.invitations(id) on delete cascade,
  guest_name text not null,
  guest_phone text default '',
  attendance_status text not null default 'attending',
  guest_count integer default 1,
  wish_message text default '',
  is_hidden boolean default false,
  ip_hash text,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- 3. Create Wishes table
create table if not exists public.wishes (
  id uuid default gen_random_uuid() primary key,
  name text not null,
  date_display text default 'ថ្មីៗនេះ',
  message text not null,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

-- 4. Create Invitation Config table
create table if not exists public.invitation_config (
  id text primary key,
  config_json jsonb not null,
  updated_at timestamp with time zone default timezone('utc'::text, now())
);

-- Enable RLS & Public Access Policies
alter table public.invitations enable row level security;
alter table public.rsvps enable row level security;
alter table public.wishes enable row level security;
alter table public.invitation_config enable row level security;

create policy "Public full access on invitations" on public.invitations for all using (true) with check (true);
create policy "Public full access on rsvps" on public.rsvps for all using (true) with check (true);
create policy "Public full access on wishes" on public.wishes for all using (true) with check (true);
create policy "Public full access on config" on public.invitation_config for all using (true) with check (true);

-- Seed the active invitation (id must match INVITATION_ID in server.py / js/supabase.js)
insert into public.invitations (id, slug, couple_names_khmer, event_type, is_active)
values ('8957c601-8044-4935-aa3d-1f369b0befa1', 'amra-wedding', 'ជា វិបុល & ស៊ុន ទេវី', 'wedding', true)
on conflict (id) do nothing;
```
